/* ══════════════════════════════════════════════════════════════════
   RESERVE & PICKUP — Local Deals Grid
   ══════════════════════════════════════════════════════════════════ */

function ReservePickupGrid() {
  const [favorites, setFavorites] = useState<Set<number>>(new Set());

  const toggleFav = (id: number) => {
    setFavorites((prev) => {
      const next = new Set(prev);
      next.has(id) ? next.delete(id) : next.add(id);
      return next;
    });
  };

  return (
    <section className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-6 lg:py-10">
      <div className="flex items-center justify-between mb-4 lg:mb-6">
        <div>
          <h2 className="text-base lg:text-xl font-bold text-text-primary flex items-center gap-2">
            Reserve &amp; Pickup
            <span className="text-[10px] font-bold bg-primary-subtle text-primary px-2 py-0.5 rounded-full">SAVE vs Online</span>
          </h2>
          <p className="text-xs lg:text-sm text-text-muted mt-0.5">Reserve at ₹10, pick up from shop — no delivery wait</p>
        </div>
        <button className="text-xs sm:text-sm font-semibold text-primary flex items-center gap-1 hover:gap-1.5 transition-all group">
          View all <ArrowRight className="w-3.5 h-3.5 group-hover:translate-x-0.5 transition-transform" />
        </button>
      </div>

      <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-4 gap-3 md:gap-4 lg:gap-5">
        {localProducts.map((product, index) => {
          const savings = product.onlinePrice - product.localPrice;
          const savingsPct = Math.round((savings / product.onlinePrice) * 100);
          const isFav = favorites.has(product.id);

          return (
            <div
              key={product.id}
              className="group bg-card-bg rounded-2xl border border-card-border overflow-hidden
                hover:shadow-lg hover:border-card-hover-border hover:-translate-y-1
                transition-all duration-300 animate-scale-in"
              style={{ animationDelay: `${index * 50}ms` }}
            >
              {/* Image area */}
              <div className="relative aspect-[4/3] bg-gradient-to-br from-surface via-surface-raised/50 to-surface flex items-center justify-center">
                <span className="text-5xl lg:text-6xl group-hover:scale-110 transition-transform duration-500 select-none">
                  {product.emoji}
                </span>

                {/* Savings badge */}
                {savingsPct > 0 && (
                  <span className="absolute top-2.5 left-2.5 text-[10px] font-extrabold text-white bg-success px-2 py-0.5 rounded-lg shadow-sm">
                    Save {savingsPct}%
                  </span>
                )}

                {/* Fav */}
                <button
                  onClick={() => toggleFav(product.id)}
                  className="absolute top-2 right-2 w-8 h-8 rounded-full bg-white/90 backdrop-blur-sm
                    flex items-center justify-center shadow-sm hover:scale-110 active:scale-90 transition-transform"
                >
                  <Heart className={`w-4 h-4 transition-colors ${isFav ? "text-red-500 fill-red-500" : "text-slate-400"}`} />
                </button>

                {/* Distance */}
                <div className="absolute bottom-2 right-2 flex items-center gap-1 bg-black/60 backdrop-blur-sm text-white text-[10px] font-semibold px-2 py-1 rounded-lg">
                  <Navigation className="w-3 h-3" />
                  {product.distance}
                </div>

                {/* Availability */}
                <div className="absolute bottom-2 left-2 flex items-center gap-1 bg-white/90 backdrop-blur-sm text-text-primary text-[10px] font-semibold px-2 py-1 rounded-lg">
